Here’s a complete, runnable, and thread-safe C# program that implements multithreaded QuickSort using the Task Parallel Library (TPL).
It uses recursion with parallel execution for large partitions and falls back to sequential sorting for small ones to avoid excessive thread overhead.

Multithreaded QuickSort in C#
Csharpusing System;
using System.Threading.Tasks;

class Program
{
    // Threshold to decide when to switch to sequential sort
    private const int PARALLEL_THRESHOLD = 1000;

    static void Main()
    {
        try
        {
            // Example array
            int[] data = { 34, 7, 23, 32, 5, 62, 32, 7, 78, 12, 45, 90, 1, 0, -5 };

            Console.WriteLine("Original array:");
            PrintArray(data);

            // Perform multithreaded quicksort
            QuickSortParallel(data, 0, data.Length - 1);

            Console.WriteLine("\nSorted array:");
            PrintArray(data);
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error: {ex.Message}");
        }
    }

    /// <summary>
    /// Multithreaded QuickSort using Task Parallel Library
    /// </summary>
    static void QuickSortParallel(int[] arr, int left, int right)
    {
        if (left >= right) return;

        int pivotIndex = Partition(arr, left, right);

        // If the partition size is large enough, sort in parallel
        if ((right - left) > PARALLEL_THRESHOLD)
        {
            Parallel.Invoke(
                () => QuickSortParallel(arr, left, pivotIndex - 1),
                () => QuickSortParallel(arr, pivotIndex + 1, right)
            );
        }
        else
        {
            // Sort sequentially for small partitions
            QuickSortParallel(arr, left, pivotIndex - 1);
            QuickSortParallel(arr, pivotIndex + 1, right);
        }
    }

    /// <summary>
    /// Partition method for QuickSort
    /// </summary>
    static int Partition(int[] arr, int left, int right)
    {
        int pivot = arr[right];
        int i = left - 1;

        for (int j = left; j < right; j++)
        {
            if (arr[j] <= pivot)
            {
                i++;
                Swap(arr, i, j);
            }
        }
        Swap(arr, i + 1, right);
        return i + 1;
    }

    /// <summary>
    /// Swap two elements in the array
    /// </summary>
    static void Swap(int[] arr, int i, int j)
    {
        if (i != j)
        {
            int temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
        }
    }

    /// <summary>
    /// Utility method to print the array
    /// </summary>
    static void PrintArray(int[] arr)
    {
        Console.WriteLine(string.Join(", ", arr));
    }
}


Key Features:

Parallel Execution: Uses Parallel.Invoke for large partitions to utilize multiple CPU cores.
Threshold Control: PARALLEL_THRESHOLD prevents overhead from creating too many threads for small partitions.
Thread-Safe: Each recursive call works on its own subarray segment, avoiding shared-state issues.
Error Handling: Wrapped in try-catch to handle unexpected runtime errors.
